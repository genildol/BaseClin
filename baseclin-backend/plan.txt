model User {
  id        Int     @id @default(autoincrement())
  name      String
  email     String  @unique
  password  String
  role      String  // admin, medico, enfermagem
  status    Boolean @default(true)
  createdAt DateTime @default(now())
  updatedAt DateTime?
  appointments Appointment[]
}

model Patient {
  id           Int       @id @default(autoincrement())
  name         String
  birthDate    DateTime
  cpf          String    @unique
  address      String?
  phone        String?
  createdAt    DateTime  @default(now())
  updatedAt    DateTime?
  appointments Appointment[]
}

model Appointment {
  id         Int       @id @default(autoincrement())
  date       DateTime
  patientId  Int
  doctorId   Int
  status     String
  notes      String?
  patient    Patient   @relation(fields: [patientId], references: [id])
  doctor     User      @relation(fields: [doctorId], references: [id])
}
